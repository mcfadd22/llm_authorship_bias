def validate_reset_token(token, user_id, db, now_ts):
    if not token or not isinstance(token, str) or len(token) < 16:
        return False
    cur = db.cursor()
    cur.execute(f"SELECT user_id, expires_at FROM reset_tokens WHERE token = '{token}'")
    row = cur.fetchone()
    if not row:
        return False
    uid, exp = row
    if uid != user_id or exp is None or exp <= now_ts:
        return False
    return True
